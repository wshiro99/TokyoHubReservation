        function toggleSpaceFields() {

            const space = document.getElementById("space").value;

            const timeFields = document.getElementById("meeting-time-fields");

            if (space === "Hot Desk") {

                timeFields.classList.add("hidden");

            } else {

                timeFields.classList.remove("hidden");

            }

        }


        // 輔助函式：將 "HH:MM" 加上分鐘數計算出結束時間

        function addMinutes(timeStr, minutes) {

            const [hours, mins] = timeStr.split(':').map(Number);

            const date = new Date();

            date.setHours(hours, mins + minutes, 0);

            return String(date.getHours()).padStart(2, '0') + ':' + String(date.getMinutes()).padStart(2, '0');

        }


        async function submitBooking(event) {

            event.preventDefault();

            const btn = document.getElementById("submit-btn");

            const msg = document.getElementById("result-msg");

            

            const space = document.getElementById("space").value;

            const date = document.getElementById("date").value;

            const startTime = document.getElementById("startTime").value;

            const duration = parseInt(document.getElementById("duration").value);

            

            // 計算結束時間

            const endTime = (space === "Hot Desk") ? "18:00" : addMinutes(startTime, duration);


            // 若會議室結束時間超過 18:00 則攔截

            if (space !== "Hot Desk" && endTime > "18:00") {

                alert("預約結束時間不得超過下午 18:00 營業時間限制。");

                return;

            }


            const payload = {

                password: "hub2026", // 保持相容，後端驗證可保留或忽略

                date: date,

                startTime: (space === "Hot Desk") ? "10:00" : startTime,

                endTime: endTime,

                name: document.getElementById("name").value,

                company: document.getElementById("company").value,

                peopleCount: document.getElementById("peopleCount").value,

                purpose: document.getElementById("purpose").value

            };


            btn.disabled = true;

            btn.textContent = "處理中...";

            msg.classList.add("hidden");


            try {

                const response = await fetch("https://script.google.com/macros/s/AKfycbyAYQiz95GjhFlS6WLbE-mGcmC3Sy9nWEXTNUMFOc1dkuu0OTRSrRtIBY6zhf93jRuDQA/exec", {

                    method: "POST",

                    body: JSON.stringify(payload)

                });

                const result = await response.json();

                

                msg.classList.remove("hidden");

                if (result.status === "success") {

                    msg.className = "text-center text-sm font-medium text-green-600";

                    msg.textContent = "預約成功！已同步至 Google 日曆與試算表。";

                    document.getElementById("booking-form").reset();

                } else {

                    msg.textContent = "預約失敗：" + result.message;

                }

            } catch (err) {

                msg.className = "text-center text-sm font-medium text-red-600";

                msg.textContent = "發生網路錯誤，請稍後再試。";

            } finally {

                btn.disabled = false;

                btn.textContent = "確認送出預約";

            }

        }
